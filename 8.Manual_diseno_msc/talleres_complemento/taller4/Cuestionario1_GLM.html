```{=html}
<script>
(function(){
  const DATA = JSON.parse(document.getElementById('qz-data').textContent);
  let idx = 0, score = 0, answered = false, opciones = [];

  // Baraja las opciones (Fisher-Yates) para que la respuesta correcta
  // no quede siempre en la misma posición
  function barajar(arr){
    const a = arr.slice();
    for(let k = a.length - 1; k > 0; k--){
      const r = Math.floor(Math.random() * (k + 1));
      [a[k], a[r]] = [a[r], a[k]];
    }
    return a;
  }
  const card = document.getElementById('qzCard');
  const fill = document.getElementById('qzFill');
  const label = document.getElementById('qzLabel');

  function render(){
    if(idx >= DATA.length){ renderResult(); return; }
    answered = false;
    const q = DATA[idx];
    opciones = barajar(q.options);
    fill.style.width = (idx/DATA.length*100) + '%';
    label.textContent = `Pregunta ${idx+1} de ${DATA.length}`;
    card.innerHTML = `
      <p class="question">${q.question}</p>
      <div class="options">
        ${opciones.map((o,i)=>`<button class="opt" data-i="${i}">${o.text}</button>`).join('')}
      </div>
      <div class="hint-row">
        <button class="hint-btn" id="qzHintBtn">Mostrar pista</button>
        <div class="hint-text" id="qzHintText" style="display:none;">${q.hint || ''}</div>
      </div>
      <div class="feedback" id="qzFeedback"></div>
      <div class="nav-row"><button class="next-btn" id="qzNextBtn">Siguiente →</button></div>
    `;
    document.getElementById('qzHintBtn').onclick = () => {
      const t = document.getElementById('qzHintText');
      t.style.display = t.style.display === 'none' ? 'block' : 'none';
    };
    card.querySelectorAll('.opt').forEach(btn=>{
      btn.onclick = () => selectOption(parseInt(btn.dataset.i));
    });
    document.getElementById('qzNextBtn').onclick = () => { idx++; render(); };
  }

  function selectOption(i){
    if(answered) return;
    answered = true;
    const q = DATA[idx];
    const opts = card.querySelectorAll('.opt');
    opts.forEach((btn,j)=>{
      btn.disabled = true;
      if(opciones[j].correct) btn.classList.add('correct');
      else if(j===i) btn.classList.add('incorrect');
      else btn.classList.add('dim');
    });
    const fb = document.getElementById('qzFeedback');
    if(opciones[i].correct){
      score++;
      fb.innerHTML = '¡Correcto! <div class="explain">' + q.explain + '</div>';
      fb.className = 'feedback show ok';
    } else {
      fb.innerHTML = 'No es correcto. <div class="explain">' + q.explain + '</div>';
      fb.className = 'feedback show bad';
    }
    document.getElementById('qzNextBtn').classList.add('show');
  }

  function renderResult(){
    fill.style.width = '100%';
    label.textContent = `${DATA.length} de ${DATA.length}`;
    const pct = Math.round(score/DATA.length*100);
    card.innerHTML = `
      <div class="result">
        <p class="score">${score}<span>/${DATA.length}</span></p>
        <p>${pct}% de aciertos. ${pct>=80?'Excelente dominio de los modelos lineales generalizados.':pct>=60?'Buen manejo general: conviene repasar los temas con errores.':'Vale la pena repasar el taller antes de la Semana 3.'}</p>
        <button class="restart-btn" id="qzRestartBtn">Intentar de nuevo</button>
      </div>
    `;
    document.getElementById('qzRestartBtn').onclick = () => { idx=0; score=0; render(); };
  }

  render();
})();
</script>
```
